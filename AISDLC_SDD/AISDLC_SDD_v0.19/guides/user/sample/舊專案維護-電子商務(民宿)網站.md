# AISDLC 詳細實戰：舊專案維護 民宿網站

**專案類型**: Brownfield (既有系統改造)
**開發環境**: Cursor AI + Claude Code
**前端技術**: React (Next.js 14) + TypeScript
**後端技術**: Spring Boot 3.x + Java 17
**資料庫**: PostgreSQL 15
**部署**: Google Cloud Platform
**適用版本**: AISDLC-SDD v0.01+
**最後更新**: 2025-12-15

> **SDD v0.01 使用者提示**：本指南基於 AISDLC-SDD v0.01。使用 AISDLC-SDD v0.01 時，請同時參考對應場景的 SDD Enhancement 文件（`scenarios/{scenario}/SDD_{SCENARIO}_ENHANCEMENT.md`），並確認所有 SCG 閘門步驟已整合到您的工作流程中。

// frontend/src/lib/websocket/useWebSocket.ts

import { useEffect, useRef, useState } from 'react';
import { Client } from '@stomp/stompjs';
import SockJS from 'sockjs-client';

export function useWebSocket(chatRoomId: string) {
  const clientRef = useRef<Client | null>(null);
  const [messages, setMessages] = useState<Message[]>([]);
  const [connected, setConnected] = useState(false);

  useEffect(() => {
    const client = new Client({
      webSocketFactory: () => new SockJS('http://localhost:8080/ws'),
      onConnect: () => {
        setConnected(true);

        // 訂閱聊天室
        client.subscribe(`/topic/chat/${chatRoomId}`, (message) => {
          const newMessage = JSON.parse(message.body);
          setMessages((prev) => [...prev, newMessage]);
        });
      },
    });

    client.activate();
    clientRef.current = client;

    return () => {
      client.deactivate();
    };
  }, [chatRoomId]);

  const sendMessage = (content: string) => {
    if (clientRef.current && connected) {
      clientRef.current.publish({
        destination: `/app/chat/${chatRoomId}`,
        body: JSON.stringify({ content }),
      });
    }
  };

  return { messages, connected, sendMessage };
}
