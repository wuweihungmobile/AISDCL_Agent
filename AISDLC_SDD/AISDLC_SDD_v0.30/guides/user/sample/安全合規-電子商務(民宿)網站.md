```java
import com.bnb.entity.Booking;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import java.time.LocalDate;
import java.util.List;

public interface BookingRepository extends JpaRepository<Booking, Long> {

    // ✅ 安全寫法：JPA Query Method
    List<Booking> findByUserId(Long userId);

        @Param("endDate") LocalDate endDate

        @Param("status") String status
```

**Claude Code 生成** (`SecurityConfig.java`):
```java
package com.bnb.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.security.web.csrf.CookieCsrfTokenRepository;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain filterChain(HttpSecurity http) throws Exception {
        http
            // ✅ 啟用 CSRF 防護
            .csrf(csrf -> csrf
                .csrfTokenRepository(CookieCsrfTokenRepository.withHttpOnlyFalse())
                .ignoringRequestMatchers("/api/public/**")  // 公開 API 排除
            )

            // ✅ Cookie 安全配置
            .sessionManagement(session -> session
                .sessionFixation().migrateSession()
            )

            // ✅ 安全 Headers
            .headers(headers -> headers
                .contentSecurityPolicy(csp ->
                    csp.policyDirectives("default-src 'self'"))
                .frameOptions(frame -> frame.deny())
                .xssProtection(xss -> xss.block(true))
            )

            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/api/public/**").permitAll()
                .requestMatchers("/api/admin/**").hasRole("ADMIN")
                .anyRequest().authenticated()
            );

        return http.build();
    }
}
```

**Claude Code 生成** (`BookingService.java`):
```java
package com.bnb.service;

import com.bnb.entity.Booking;
import com.bnb.exception.UnauthorizedException;
import org.springframework.security.access.prepost.PreAuthorize;
import org.springframework.security.core.Authentication;
import org.springframework.stereotype.Service;

@Service
public class BookingService {

    // ✅ 方法級權限檢查
    @PreAuthorize("hasRole('USER')")
    public Booking createBooking(CreateBookingRequest request, Authentication auth) {
        Long userId = getUserIdFromAuth(auth);

        Booking booking = Booking.builder()
            .userId(userId)
            .roomId(request.getRoomId())
            .checkIn(request.getCheckIn())
            .checkOut(request.getCheckOut())
            .build();

        return bookingRepository.save(booking);
    }

    // ✅ 資源擁有者檢查
    @PreAuthorize("hasRole('USER')")
    public Booking updateBooking(Long bookingId, UpdateBookingRequest request, Authentication auth) {
        Booking booking = bookingRepository.findById(bookingId)
            .orElseThrow(() -> new NotFoundException("訂單不存在"));

        Long userId = getUserIdFromAuth(auth);

        // ✅ 檢查資源擁有者
        if (!booking.getUserId().equals(userId)) {
            throw new UnauthorizedException("您無權修改此訂單");
        }

        booking.setCheckIn(request.getCheckIn());
        booking.setCheckOut(request.getCheckOut());

        return bookingRepository.save(booking);
    }

    // ✅ 管理員可查看所有訂單
    @PreAuthorize("hasRole('ADMIN')")
    public List<Booking> getAllBookings() {
        return bookingRepository.findAll();
    }

    private Long getUserIdFromAuth(Authentication auth) {
        UserDetails userDetails = (UserDetails) auth.getPrincipal();
        return userDetails.getUserId();
    }
}
```

**Claude Code 生成** (`UserService.java`):
```java
package com.bnb.service;

import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

@Service
public class UserService {

    private final BCryptPasswordEncoder passwordEncoder = new BCryptPasswordEncoder(12);  // ✅ 12 rounds

    public User register(RegisterRequest request) {
        // ✅ 密碼強度驗證
        validatePasswordStrength(request.getPassword());

        // ✅ BCrypt 加密（自動加鹽）
        String hashedPassword = passwordEncoder.encode(request.getPassword());

        User user = User.builder()
            .email(request.getEmail())
            .password(hashedPassword)  // ✅ 儲存加密後的密碼
            .build();

        return userRepository.save(user);
    }

    public boolean authenticate(String email, String rawPassword) {
        User user = userRepository.findByEmail(email)
            .orElseThrow(() -> new NotFoundException("使用者不存在"));

        // ✅ BCrypt 驗證
        return passwordEncoder.matches(rawPassword, user.getPassword());
    }

    private void validatePasswordStrength(String password) {
        // ✅ 密碼強度規則：8+ 字元、大小寫、數字、特殊字元
        String passwordRegex = "^(?=.*[a-z])(?=.*[A-Z])(?=.*\\d)(?=.*[@$!%*?&])[A-Za-z\\d@$!%*?&]{8,}$";
        if (!password.matches(passwordRegex)) {
            throw new ValidationException("密碼必須包含至少 8 字元、大小寫字母、數字和特殊字元");
        }
    }
}
```

```typescript
'use client'

import { useState, useEffect } from 'react'
import Cookies from 'js-cookie'

export default function CookieConsent() {
  const [showBanner, setShowBanner] = useState(false)

  useEffect(() => {
    const consent = Cookies.get('cookie-consent')
    if (!consent) {
      setShowBanner(true)
    }
  }, [])

  const acceptCookies = () => {
    Cookies.set('cookie-consent', 'accepted', { expires: 365, sameSite: 'strict' })
    setShowBanner(false)
  }

  if (!showBanner) return null

  return (
    <div className="fixed bottom-0 left-0 right-0 bg-gray-900 text-white p-4 z-50">
      <div className="container mx-auto flex justify-between items-center">
        <p>
          我們使用 Cookie 來改善您的使用體驗。繼續使用本網站即表示您同意我們的{' '}
          <a href="/privacy-policy" className="underline">隱私政策</a>。
        </p>
        <button onClick={acceptCookies} className="bg-blue-600 px-6 py-2 rounded">
          接受
        </button>
      </div>
    </div>
  )
}
```

```java
package com.bnb.controller;

import com.bnb.service.GdprService;
import org.springframework.security.core.Authentication;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/api/gdpr")
public class GdprController {

    private final GdprService gdprService;

    // ✅ Right to Access：匯出使用者資料
    @GetMapping("/export-data")
    public ResponseEntity<byte[]> exportUserData(Authentication auth) {
        Long userId = getUserIdFromAuth(auth);

        // 匯出為 JSON
        byte[] jsonData = gdprService.exportUserData(userId);

        return ResponseEntity.ok()
            .header("Content-Disposition", "attachment; filename=user-data.json")
            .contentType(MediaType.APPLICATION_JSON)
            .body(jsonData);
    }

    // ✅ Right to be Forgotten：刪除使用者資料
    @DeleteMapping("/delete-account")
    public ResponseEntity<Void> deleteAccount(
        @RequestParam String confirmationToken,
        Authentication auth
    ) {
        Long userId = getUserIdFromAuth(auth);

        // 二次確認（需先發送確認 Email）
        gdprService.deleteUserAccount(userId, confirmationToken);

        return ResponseEntity.noContent().build();
    }
}
```
